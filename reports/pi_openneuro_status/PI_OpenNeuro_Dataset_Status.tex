\section*{Executive summary (1 minute)}

\begin{itemize}[leftmargin=1.2em,itemsep=2pt]
\item The \textbf{research BIDS dataset is structurally complete}: \textbf{84 participants}, \textbf{124 sessions}, \textbf{7{,}956} imaging NIfTI (+ JSON).
\item \textbf{BIDS validator: 0 errors} (documentable warnings remain: missing events for some runs, longitudinal incompleteness, placeholder authors).
\item \textbf{Defacing is ready} for public release (T1w 356/356); pre-flight anatomical replacement plan \textbf{PASS} (481/481).
\item \textbf{Not ready to upload to OpenNeuro today.} Two hard blockers remain:
  \begin{enumerate}[leftmargin=1.4em,itemsep=1pt]
  \item Finish the \textbf{defaced public release package} (currently $\sim$68/84 subjects; anatomicals not yet swapped in).
  \item Replace \textbf{placeholder Authors} and set \textbf{License: CC0}.
  \end{enumerate}
\item \textbf{MRIQC} is useful but incomplete (78\% sessions fully covered). Either finish the AFNI patch re-run or disclose coverage in Technical Validation --- this is a caution, not an OpenNeuro packaging blocker.
\end{itemize}

\noindent\textit{Bottom line for decision:} science package and privacy pipeline are largely done; remaining work is packaging + authorship/license + MRIQC disclosure/completion.

\subsection{MRIQC}
\begin{itemize}[leftmargin=1.2em,itemsep=1pt]
\item T1w IQMs: \textbf{353/356 (99.2\%)}
\item BOLD magnitude IQMs: \textbf{1{,}367/1{,}496 (91.4\%)}
\item Fully complete sessions: \textbf{97/124 (78.2\%)}

\end{itemize}

\subsection{DWI QC}
\begin{itemize}[leftmargin=1.2em,itemsep=1pt]

\item Publication recommendation: document REVIEW observations; \textbf{not release-blocking}.
